```vue
  </el-table-column>
  <el-table-column label="数量">
    <el-input-number v-model="row.quantity" @change="calculateAmount" />
  </el-table-column>

```

### 2. 实时金额计算
```typescript
const orderTotal = computed(() => {
  return form.items.reduce((sum, item) => sum + (item.amount || 0), 0)
})
```

### 3. 状态流转控制
```vue

<el-button v-if="row.status === 'PENDING'" @click="handleApprove">审核</el-button>
```
